\begin{enumerate}[start = 1, itemsep=1.5em]
    \item Evaluate $\sum\limits_{k\in S}\frac{1}{k^3}$, where $S$ is the set of positive integers such that $\frac{1}{k}$ has a terminating decimal expansion.
    \item Show that every positive integer is a sum of one or more numbers of the form $2^r3^s$, where $r$ and $s$ are nonnegative integers and no summand divides the other. For example, $23 = 9+8+6$.  
    \item Find all positive integers less than 10000 which are palindromic in both base 10 and base 11.
    \item A chord of constant length slides around in a semicircle. The midpoint of the chord and the projections of its ends upon the base form the vertices of a triangle. Prove that the triangle is isoscles and never changes its shape.
    \item Prove that for any positive real numbers $a,b,c$,\begin{center}
        $\frac{a^2}{b}+\frac{b^2}{c}+\frac{c^2}{a}+\sqrt{ab}+\sqrt{bc}+\sqrt{ca}\geq 2(a+b+c)$.
    \end{center}
    \item Find all triples $(p,q,r)$ of primes such that \begin{center}
        $p^2 + 2q^2 + r^2=3pqr$.
    \end{center}
\end{enumerate}
